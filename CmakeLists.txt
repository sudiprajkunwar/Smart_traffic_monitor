cmake_minimum_required(VERSION 3.10)

# Project name and C++ standard
project(SmartTrafficMonitor)
set(CMAKE_CXX_STANDARD 20)

# Include directories
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/src)

# Collect source files
file(GLOB_RECURSE SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")

# Find OpenCV package
find_package(OpenCV REQUIRED)
if(OpenCV_FOUND)
    include_directories(${OpenCV_INCLUDE_DIRS})
else()
    message(FATAL_ERROR "OpenCV not found")
endif()

# Use pkg-config to find Tesseract
find_package(PkgConfig REQUIRED)
pkg_check_modules(TESSERACT REQUIRED tesseract)

if(TESSERACT_FOUND)
    include_directories(${TESSERACT_INCLUDE_DIRS})
    link_directories(${TESSERACT_LIBRARY_DIRS})
else()
    message(FATAL_ERROR "Tesseract not found")
endif()

# Add executable target
add_executable(SmartTrafficMonitor ${SOURCES})

# Link libraries
target_link_libraries(SmartTrafficMonitor ${OpenCV_LIBS} ${TESSERACT_LIBRARIES})
