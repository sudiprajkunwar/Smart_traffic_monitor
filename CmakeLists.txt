cmake_minimum_required(VERSION 3.10)

# Project name and C++ standard
project(SmartTrafficMonitor)
set(CMAKE_CXX_STANDARD 20)

# Include directories
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/src)

# Collect source files
file(GLOB_RECURSE SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")

# Find OpenCV package
find_package(OpenCV REQUIRED)
if(OpenCV_FOUND)
    include_directories(${OpenCV_INCLUDE_DIRS})
else()
    message(FATAL_ERROR "OpenCV not found")
endif()

# Use pkg-config to find Tesseract
find_package(PkgConfig REQUIRED)
pkg_check_modules(TESSERACT REQUIRED tesseract)

if(TESSERACT_FOUND)
    include_directories(${TESSERACT_INCLUDE_DIRS})
    link_directories(${TESSERACT_LIBRARY_DIRS})
else()
    message(FATAL_ERROR "Tesseract not found")
endif()

# Find Google Test
find_package(GTest REQUIRED)
if(GTest_FOUND)
    include_directories(${GTEST_INCLUDE_DIRS})

else()
    message(FATAL_ERROR "Google Test not found")
endif()

# Add executable target for the main application
add_executable(SmartTrafficMonitor ${SOURCES})

# Link libraries for the main application
target_link_libraries(SmartTrafficMonitor ${OpenCV_LIBS} ${TESSERACT_LIBRARIES})

# Collect test source files
file(GLOB_RECURSE TEST_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp")

# Collect all source files needed for the test executable
file(GLOB_RECURSE TEST_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")


# Remove the main application file (if any) from the test dependencies
list(REMOVE_ITEM TEST_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")

# Add executable target for tests
add_executable(SmartTrafficMonitorTests ${TEST_SOURCES} ${TEST_DEPS})

# Link libraries for the test executable
target_link_libraries(SmartTrafficMonitorTests ${OpenCV_LIBS} ${TESSERACT_LIBRARIES} GTest::GTest GTest::Main)

# Add a test
add_test(NAME SmartTrafficMonitorTests COMMAND SmartTrafficMonitorTests)
